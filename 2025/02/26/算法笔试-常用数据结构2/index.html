1441. 用栈操作构建数组

func buildArray(target []int, n int) []string {
    ans := make([]string, 0) 
    for i, j := 1, 0; j < len(target) && i <= n; i++ {
        if i == target[j] {
            ans = append(ans, "Push")
            j++
        } else {
            ans = append(ans, "Push")
            ans = append(ans, "Pop")
        }
    }
    return ans
}

844. 比较含退格的字符串

func backspaceCompare(s string, t string) bool {
    helper := func (a string) string {
        stk := make([]byte, 0)
        for _, v := range []byte(a) {
            if v != '#' {
                stk = append(stk, v)
            } else if len(stk) > 0 {
                stk = stk[:len(stk)-1]
            }
        }
        return string(stk)
    }
    return helper(s) == helper(t)
}
